# ---------- Environment ----------
# HTTP client. Used to call the model API.
httpx>=0.27
# Read variables from .env so secrets stay out of the code.
python-dotenv>=1.0

# ---------- Service ----------
# Web framework. Turns our code into an HTTP service.
fastapi>=0.115
# The server that actually runs FastAPI.
uvicorn[standard]>=0.32
# Data validation. FastAPI relies on it for request bodies.
pydantic>=2.9

# ---------- RAG ----------
# Vector math. Embeddings are stored as matrices, so similarity search
# becomes a single matrix multiplication.
numpy>=1.26
# Chinese word segmentation, used by BM25 keyword retrieval.
# Pure Python, no model download required.
jieba>=0.42

# ---------- Agent orchestration ----------
# Agent orchestration as a state graph. Optional in the sense that
# app/agents/react.py implements the same loop by hand with zero
# dependencies -- but installed so the two can be compared side by side.
# Pulls in langchain-core, langgraph-checkpoint, langsmith, etc.
langgraph>=0.2

# ---------- SSH ----------
# SSH client library, used ONLY for password-authenticated hosts.
# Key auth still goes through the `ssh` CLI (already installed, and
# BatchMode=yes keeps automation from hanging on a prompt).
# A password cannot be fed to the CLI non-interactively on Windows:
# the switch that forces askpass (SSH_ASKPASS_REQUIRE) does not exist
# in Win32-OpenSSH, and sshpass is not available everywhere.
# A library has no "waiting for input" state to hang in.
paramiko>=3.4

# ---------- MCP ----------
# Official Model Context Protocol SDK. Exposes the 6 tools in
# app/tools/ops.py to any MCP client (Cursor, Claude Desktop, ...).
# NOTE: v2 renamed FastMCP to MCPServer --
#       from mcp.server.mcpserver import MCPServer
#       and the schema field is input_schema, not inputSchema.
mcp>=2.0
